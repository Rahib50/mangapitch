
-- MangaPitch Database Schema
-- Import via XAMPP phpMyAdmin or: mysql -u root -p < schema.sql

CREATE DATABASE IF NOT EXISTS mangapitch;
USE mangapitch;

CREATE TABLE Users (
    UserID        INT AUTO_INCREMENT PRIMARY KEY,
    Name          VARCHAR(100)  NOT NULL,
    Email         VARCHAR(150)  NOT NULL UNIQUE,
    Password      VARCHAR(255)  NOT NULL,          -- store bcrypt hash
    Role          ENUM('Mangaka', 'Studio', 'Admin') NOT NULL,
    CreatedAt     TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Mangaka (
    UserID        INT PRIMARY KEY,
    PortfolioLink VARCHAR(255),
    FOREIGN KEY (UserID) REFERENCES Users(UserID) ON DELETE CASCADE
);

CREATE TABLE Studio (
    UserID             INT PRIMARY KEY,
    RegistrationNumber VARCHAR(100),
    FOREIGN KEY (UserID) REFERENCES Users(UserID) ON DELETE CASCADE
);

CREATE TABLE Admin (
    UserID       INT PRIMARY KEY,
    AdminLevel   TINYINT DEFAULT 1,
    FOREIGN KEY (UserID) REFERENCES Users(UserID) ON DELETE CASCADE
);

CREATE TABLE Connections (
    UserID_A  INT NOT NULL,
    UserID_B  INT NOT NULL,
    PRIMARY KEY (UserID_A, UserID_B),
    FOREIGN KEY (UserID_A) REFERENCES Users(UserID) ON DELETE CASCADE,
    FOREIGN KEY (UserID_B) REFERENCES Users(UserID) ON DELETE CASCADE
);

CREATE TABLE Manga (
    MangaID      INT AUTO_INCREMENT PRIMARY KEY,
    MangakaID    INT NOT NULL,
    Title        VARCHAR(200) NOT NULL,
    Synopsis     TEXT,
    PublishDate  DATE,
    CreatedAt    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (MangakaID) REFERENCES Mangaka(UserID) ON DELETE CASCADE
);

CREATE TABLE Genre (
    GenreID   INT AUTO_INCREMENT PRIMARY KEY,
    GenreName VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE Manga_Genre_Map (
    MangaID  INT NOT NULL,
    GenreID  INT NOT NULL,
    PRIMARY KEY (MangaID, GenreID),
    FOREIGN KEY (MangaID) REFERENCES Manga(MangaID) ON DELETE CASCADE,
    FOREIGN KEY (GenreID) REFERENCES Genre(GenreID) ON DELETE CASCADE
);

CREATE TABLE Analytics (
    AnalyticsID  INT AUTO_INCREMENT PRIMARY KEY,
    MangaID      INT NOT NULL UNIQUE,
    TotalViews   INT DEFAULT 0,
    VolumesSold  INT DEFAULT 0,
    FOREIGN KEY (MangaID) REFERENCES Manga(MangaID) ON DELETE CASCADE
);

CREATE TABLE Bids (
    BidID      INT AUTO_INCREMENT PRIMARY KEY,
    MangaID    INT NOT NULL,
    StudioID   INT NOT NULL,
    BidAmount  DECIMAL(12, 2) NOT NULL,
    Status     ENUM('Pending', 'Accepted', 'Rejected', 'Withdrawn') DEFAULT 'Pending',
    CreatedAt  TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (MangaID)  REFERENCES Manga(MangaID) ON DELETE CASCADE,
    FOREIGN KEY (StudioID) REFERENCES Studio(UserID) ON DELETE CASCADE
);

CREATE TABLE Contracts (
    ContractID    INT AUTO_INCREMENT PRIMARY KEY,
    BidID         INT NOT NULL UNIQUE,
    SignedDate    DATE,
    ProductStatus ENUM('Pre-Production', 'In-Production', 'Post-Production', 'Released') DEFAULT 'Pre-Production',
    FOREIGN KEY (BidID) REFERENCES Bids(BidID) ON DELETE CASCADE
);

CREATE TABLE Messages (
    MessageID    INT AUTO_INCREMENT PRIMARY KEY,
    SenderID     INT NOT NULL,
    ReceiverID   INT NOT NULL,
    MessageText  TEXT NOT NULL,
    Timestamp    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (SenderID)   REFERENCES Users(UserID) ON DELETE CASCADE,
    FOREIGN KEY (ReceiverID) REFERENCES Users(UserID) ON DELETE CASCADE
);

INSERT INTO Genre (GenreName) VALUES
